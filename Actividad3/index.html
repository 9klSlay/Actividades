Ejemplo de If, Else If y Else

// ==================== // Caso 1: Todas son falsas // ==================== if (5 > 10) { console.log("Primera condición verdadera"); } else if (2 > 8) { console.log("Segunda condición verdadera"); } else { console.log("Todas las expresiones son falsas"); } // ==================== // Caso 2: Todas son verdaderas // ==================== if (10 > 5) { console.log("Primera condición verdadera"); } else if (8 > 3) { console.log("Segunda condición verdadera"); } else { console.log("Todas las expresiones son falsas"); } // ==================== // Caso 3: Múltiples expresiones verdaderas // ==================== if (9154 > 65) { console.log("Primera condición verdadera"); } else if (9251 > 2561) { console.log("Segunda condición verdadera"); } else { console.log("Todas las expresiones son falsas"); }
